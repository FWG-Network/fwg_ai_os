# ── Stage 1: Build dependencies ────────────────────────────────────────────────
FROM python:3.11.9-slim AS builder

WORKDIR /build

# Install build-only system deps
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    libpq-dev \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Copy and install Python dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir --upgrade pip==24.0 \
    && pip install --no-cache-dir --prefix=/install -r requirements.txt

# ── Stage 2: Production runtime ────────────────────────────────────────────────
FROM python:3.11.9-slim AS runtime

LABEL org.opencontainers.image.title="FWG Media Engine — Backend"
LABEL org.opencontainers.image.vendor="FWG Media"
LABEL org.opencontainers.image.schema-version="1.0"

# Runtime system deps only
RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq5 \
    curl \
    tini \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# Create non-root user
RUN groupadd --gid 1001 fwg \
    && useradd --uid 1001 --gid fwg --no-create-home --shell /bin/false fwg

WORKDIR /app

# Copy installed packages from builder
COPY --from=builder /install /usr/local

# Copy application source (NEVER copy fwg_transformer.py here — it stays on host mount)
COPY --chown=fwg:fwg backend/ ./backend/
COPY --chown=fwg:fwg alembic/ ./alembic/
COPY --chown=fwg:fwg alembic.ini .

# Set permissions
RUN chmod -R 550 /app/backend \
    && chmod 440 /app/alembic.ini

# Switch to non-root
USER fwg

# Expose port
EXPOSE 8000

# Health check
HEALTHCHECK --interval=15s --timeout=5s --start-period=30s --retries=3 \
    CMD curl -sf http://localhost:8000/health/live || exit 1

# Metadata
ARG APP_VERSION=dev
ARG BUILD_SHA=unknown
ENV APP_VERSION=${APP_VERSION} \
    BUILD_SHA=${BUILD_SHA} \
    PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PYTHONFAULTHANDLER=1

# Use tini as PID 1 to handle signals properly
ENTRYPOINT ["/usr/bin/tini", "--"]

# Production: uvicorn with graceful shutdown
CMD ["uvicorn", "backend.main:app", \
    "--host", "0.0.0.0", \
    "--port", "8000", \
    "--workers", "4", \
    "--worker-class", "uvicorn.workers.UvicornWorker", \
    "--timeout-keep-alive", "30", \
    "--access-log", \
    "--log-level", "info"]
